   198→    private fun registerUsbMonitor() {
   199→        if (isRegistered) {
   200→            Log.i(TAG, "USBMonitor already registered, skipping")
   201→            return
   202→        }
   203→
   204→        try {
   205→            if (usbMonitor == null) {
   206→                usbMonitor = USBMonitor(context, object : OnDeviceConnectListener {
   207→                    override fun onAttach(device: UsbDevice) {
   208→                        Log.i(TAG, "═══════════════════════════════════════════════════")
   209→                        Log.i(TAG, "USB ATTACHED: ${device.deviceName}, vid=0x${Integer.toHexString(device.vendorId)}, pid=0x${Integer.toHexString(device.productId)}")
   210→                        Log.i(TAG, "═══════════════════════════════════════════════════")
   211→
   212→                        if (isUVCDevice(device)) {
   213→                            Log.i(TAG, "★ UVC VIDEO DEVICE detected! Requesting permission...")
   214→                            discoveredDevices[device.deviceName] = device
   215→                            updateAvailableCameras()
   216→                            requestUsbPermission(device)
   217→                        } else {
   218→                            Log.i(TAG, "Not a UVC device, ignoring")
   219→                        }
   220→                    }
   221→
   222→                    override fun onDisconnect(device: UsbDevice, ctrlBlock: UsbControlBlock) {
   223→                        Log.i(TAG, "USB DISCONNECTED: ${device.deviceName}")
   224→                        handleDeviceDisconnect(device)
   225→                    }
   226→
   227→                    override fun onConnect(device: UsbDevice, ctrlBlock: UsbControlBlock, createNew: Boolean) {
   228→                        Log.i(TAG, "═══════════════════════════════════════════════════")
   229→                        Log.i(TAG, "USB CONNECTED (permission granted): ${device.deviceName}")
   230→                        Log.i(TAG, "  createNew=$createNew")
   231→                        Log.i(TAG, "═══════════════════════════════════════════════════")
   232→
   233→                        currentUsbDevice = device
   234→                        currentCtrlBlock = ctrlBlock
   235→
   236→                        // Open UVC camera on background thread
   237→                        backgroundHandler?.post {
   238→                            openUVCCamera(device, ctrlBlock)
   239→                        }
   240→                    }
   241→
   242→                    // NOTE: onDettach (double-t) is the correct spelling in this library
   243→                    override fun onDettach(device: UsbDevice) {
   244→                        Log.i(TAG, "USB DETACHED: ${device.deviceName}")
   245→                        discoveredDevices.remove(device.deviceName)
   246→                        updateAvailableCameras()
   247→                        handleDeviceDisconnect(device)
   248→                    }
   249→
   250→                    override fun onCancel(device: UsbDevice) {
   251→                        Log.i(TAG, "USB permission CANCELLED for: ${device.deviceName}")
   252→                    }
   253→                })
   254→            }
   255→
   256→            usbMonitor?.register()
   257→            isRegistered = true
   258→            Log.i(TAG, "USBMonitor registered successfully")
   259→
   260→            // Register BroadcastReceiver for USB permission results
   261→            registerUsbPermissionReceiver()
   262→
   263→            // Enumerate already-connected USB devices
   264→            enumerateConnectedDevices()
   265→
   266→            // ── Android 14+ USB polling fallback ──
   267→            // The USBMonitor library may fail to register its internal
   268→            // BroadcastReceiver on Android 14+ (SecurityException due to
   269→            // missing RECEIVER_EXPORTED flag). This polling mechanism
   270→            // detects USB devices as a fallback by checking UsbManager.deviceList
   271→            // every 2 seconds. It's a no-op if USBMonitor is working correctly
   272→            // (devices already discovered are skipped).
   273→            startUsbPolling()
   274→
   275→        } catch (e: Exception) {
   276→            Log.e(TAG, "Failed to register USBMonitor: ${e.message}", e)
   277→        }
   278→    }

   322→    // ═══════════════════════════════════════════════════════════════
   323→    // Android 14+ USB Polling Fallback
   324→    // ═══════════════════════════════════════════════════════════════
   325→    // On Android 14 (API 34), the USBMonitor library's internal
   326→    // registerReceiver() may throw SecurityException due to missing
   327→    // RECEIVER_EXPORTED/RECEIVER_NOT_EXPORTED flag. This causes
   328→    // USB device detection to silently fail — no onAttach callback,
   329→    // no permission dialog, no camera detected.
   330→    //
   331→    // This polling mechanism checks UsbManager.deviceList every 2
   332→    // seconds as a fallback. It's a no-op if USBMonitor is working
   333→    // correctly (devices already discovered are skipped).
   334→    // ═══════════════════════════════════════════════════════════════
   335→
   336→    private fun startUsbPolling() {
   337→        stopUsbPolling() // Clear any existing polling
   338→        usbPollingRunnable = object : Runnable {
   339→            override fun run() {
   340→                if (isDestroying) return
   341→                try {
   342→                    pollUsbDevices()
   343→                } catch (e: Exception) {
   344→                    Log.e(TAG, "USB polling error: ${e.message}")
   345→                }
   346→                // Schedule next poll
   347→                usbPollingRunnable?.let {
   348→                    mainHandler.postDelayed(it, USB_POLLING_INTERVAL_MS)
   349→                }
   350→            }
   351→        }
   352→        // Start polling after a short delay
   353→        usbPollingRunnable?.let { mainHandler.postDelayed(it, USB_POLLING_INTERVAL_MS) }
   354→        Log.i(TAG, "USB polling fallback started (every ${USB_POLLING_INTERVAL_MS}ms)")
   355→    }
   356→
   357→    private fun stopUsbPolling() {
   358→        usbPollingRunnable?.let { mainHandler.removeCallbacks(it) }
   359→        usbPollingRunnable = null
   360→    }

   362→    /**
   363→     * Poll UsbManager.deviceList for new UVC devices.
   364→     * This is a fallback for Android 14+ where USBMonitor's BroadcastReceiver
   365→     * may fail to register. It's idempotent — skips already-discovered devices.
   366→     */
   367→    private fun pollUsbDevices() {
   368→        try {
   369→            val usbManager = context.getSystemService(Context.USB_SERVICE) as? UsbManager ?: return
   370→            var newDeviceFound = false
   371→
   372→            for (device in usbManager.deviceList.values) {
   373→                if (discoveredDevices.containsKey(device.deviceName)) continue // Already known
   374→
   375→                if (isUVCDevice(device)) {
   376→                    Log.i(TAG, "★ Polling found NEW UVC device: ${device.deviceName} vid=0x${Integer.toHexString(device.vendorId)}")
   377→                    discoveredDevices[device.deviceName] = device
   378→                    newDeviceFound = true
   379→
   380→                    // Request permission if not already granted
   381→                    if (!usbManager.hasPermission(device)) {
   382→                        requestUsbPermission(device)
   383→                    } else {
   384→                        // Permission already granted — try to open the camera
   385→                        // The USBMonitor's onConnect won't fire if it's broken,
   386→                        // so we need to manually trigger the camera open
   387→                        tryOpenUVCCameraWithPermission(device)
   388→                    }
   389→                }
   390→            }
   391→
   392→            // Check for disconnected devices (plugged out since last poll)
   393→            val connectedNames = usbManager.deviceList.keys
   394→            val removedKeys = discoveredDevices.keys.filter { it !in connectedNames }
   395→            for (key in removedKeys) {
   396→                val device = discoveredDevices.remove(key)
   397→                if (device != null) {
   398→                    Log.i(TAG, "Polling detected USB DISCONNECT: ${device.deviceName}")
   399→                    handleDeviceDisconnect(device)
   400→                    newDeviceFound = true
   401→                }
   402→            }
   403→
   404→            if (newDeviceFound) {
   405→                updateAvailableCameras()
   406→            }
   407→        } catch (e: Exception) {
   408→            Log.e(TAG, "USB polling failed: ${e.message}")
   409→        }
   410→    }

   412→    /**
   413→     * Try to open a UVC camera when permission is already granted.
   414→     * This is needed when the USBMonitor's onConnect callback doesn't fire
   415→     * (e.g., on Android 14 where the library's BroadcastReceiver fails).
   416→     */
   417→    private fun tryOpenUVCCameraWithPermission(device: UsbDevice) {
   418→        try {
   419→            val usbManager = context.getSystemService(Context.USB_SERVICE) as? UsbManager ?: return
   420→            if (!usbManager.hasPermission(device)) return
   421→
   422→            // On Android 14+, USBMonitor's onConnect may never fire because
   423→            // the library's internal registerReceiver() throws SecurityException.
   424→            // Try direct open first (bypasses USBMonitor entirely).
   425→            tryOpenUVCCameraDirect(device)
   426→        } catch (e: Exception) {
   427→            Log.e(TAG, "tryOpenUVCCameraWithPermission error: ${e.message}")
   428→        }
   429→    }
   430→
   431→    /**
   432→     * Directly open a UVC camera when USB permission is already granted.
   433→     * On Android 14+, USBMonitor's internal BroadcastReceiver may fail due to
   434→     * missing RECEIVER_EXPORTED flag. This method uses USBMonitor.requestPermission()
   435→     * on an already-permitted device, which internally calls processConnect() → onConnect()
   436→     * WITHOUT needing the BroadcastReceiver. This is the key workaround for Android 14.
   437→     */
   438→    private fun tryOpenUVCCameraDirect(device: UsbDevice) {
   439→        try {
   440→            val usbManager = context.getSystemService(Context.USB_SERVICE) as? UsbManager ?: return
   441→            if (!usbManager.hasPermission(device)) return
   442→
   443→            // Don't open if already connected to this device
   444→            if (currentUsbDevice?.deviceName == device.deviceName && _isConnected.value) {
   445→                Log.i(TAG, "tryOpenUVCCameraDirect: already connected to ${device.deviceName}")
   446→                return
   447→            }
   448→
   449→            val monitor = usbMonitor
   450→            if (monitor != null) {
   451→                // KEY INSIGHT: USBMonitor.requestPermission() on an already-permitted device
   452→                // calls processConnect() internally, which creates a UsbControlBlock and
   453→                // triggers onConnect(). This works even when the BroadcastReceiver is broken
   454→                // on Android 14 because processConnect() doesn't need the BroadcastReceiver.
   455→                Log.i(TAG, "tryOpenUVCCameraDirect: calling monitor.requestPermission() for already-permitted device")
   456→                backgroundHandler?.post {
   457→                    try {
   458→                        monitor.requestPermission(device)
   459→                    } catch (e: Exception) {
   460→                        Log.e(TAG, "USBMonitor.requestPermission() failed on already-permitted device: ${e.message}")
   461→                        // Last resort: try to force open via UsbManager.openDevice()
   462→                        tryForceOpenViaUsbManager(device)
   463→                    }
   464→                }
   465→            } else {
   466→                Log.e(TAG, "tryOpenUVCCameraDirect: No USBMonitor available")
   467→                tryForceOpenViaUsbManager(device)
   468→            }
   469→        } catch (e: Exception) {
   470→            Log.e(TAG, "tryOpenUVCCameraDirect error: ${e.message}")
   471→        }
   472→    }
   473→
   474→    /**
   475→     * Last-resort fallback: try to open camera by registering a fresh USBMonitor
   476→     * and requesting permission. This is for the case where USBMonitor is completely
   477→     * broken on Android 14 MIUI.
   478→     */
   479→    private fun tryForceOpenViaUsbManager(device: UsbDevice) {
   480→        try {
   481→            val usbManager = context.getSystemService(Context.USB_SERVICE) as? UsbManager ?: return
   482→            if (!usbManager.hasPermission(device)) return
   483→
   484→            // Don't open if already connected to this device
   485→            if (currentUsbDevice?.deviceName == device.deviceName && _isConnected.value) {
   486→                return
   487→            }
   488→
   489→            Log.i(TAG, "tryForceOpenViaUsbManager: Attempting re-registration of USBMonitor")
   490→            // Try re-registering the USBMonitor — this might work on second attempt
   491→            try {
   492→                unregisterUsbMonitor()
   493→                isRegistered = false
   494→                registerUsbMonitor()
   495→            } catch (e: Exception) {
   496→                Log.e(TAG, "Re-registration failed: ${e.message}")
   497→            }
   498→        } catch (e: Exception) {
   499→            Log.e(TAG, "tryForceOpenViaUsbManager error: ${e.message}")
   500→        }
   501→    }

   504→    // USB Permission — Required for UVC device access
   505→    // ═══════════════════════════════════════════════════════════════
   506→
   507→    private fun registerUsbPermissionReceiver() {
   508→        if (usbPermissionReceiver != null) return
   509→
   510→        usbPermissionReceiver = object : BroadcastReceiver() {
   511→            override fun onReceive(ctx: Context, intent: Intent) {
   512→                if (ACTION_USB_PERMISSION == intent.action) {
   513→                    synchronized(this) {
   514→                        @Suppress("DEPRECATION")
   515→                        val device = intent.getParcelableExtra<UsbDevice>(UsbManager.EXTRA_DEVICE)
   516→                        val granted = intent.getBooleanExtra(UsbManager.EXTRA_PERMISSION_GRANTED, false)
   517→
   518→                        Log.i(TAG, "USB permission result: device=${device?.deviceName}, granted=$granted")
   519→
   520→                        if (granted && device != null) {
   521→                            Log.i(TAG, "USB permission GRANTED for ${device.deviceName}")
   522→                            // BUG FIX: On Android 14+, USBMonitor's onConnect may never fire
   523→                            // because the library's internal registerReceiver() throws
   524→                            // SecurityException. We MUST open the camera directly here.
   525→                            tryOpenUVCCameraDirect(device)
   526→                        } else if (device != null) {
   527→                            Log.w(TAG, "USB permission DENIED for ${device.deviceName}")
   528→                        } else {
   529→                            // No device in intent — try to find device from polling
   530→                            Log.w(TAG, "USB permission result: no device in intent extras")
   531→                        }
   532→                    }
   533→                }
   534→            }
   535→        }
   536→
   537→        val filter = IntentFilter(ACTION_USB_PERMISSION)
   538→        // ── Android 14+ (API 34) requires RECEIVER_EXPORTED or RECEIVER_NOT_EXPORTED ──
   539→        // Without this flag, registerReceiver() throws SecurityException on Android 14+,
   540→        // which silently prevents USB permission dialogs from working.
   541→        // This receiver is NOT exported (only receives our own PendingIntent broadcasts).
   542→        if (android.os.Build.VERSION.SDK_INT >= android.os.Build.VERSION_CODES.TIRAMISU) {
   543→            context.registerReceiver(usbPermissionReceiver, filter, android.content.Context.RECEIVER_NOT_EXPORTED)
   544→        } else {
   545→            context.registerReceiver(usbPermissionReceiver, filter)
   546→        }
   547→        Log.i(TAG, "USB permission BroadcastReceiver registered")
   548→    }

   561→    private fun isUVCDevice(device: UsbDevice): Boolean {
   562→        // Check by interface class (UVC = Video class)
   563→        for (i in 0 until device.interfaceCount) {
   564→            val iface = device.getInterface(i)
   565→            // USB Video Class: Class 14 (0x0E) or Class 239 (0xEF) Subclass 2
   566→            if (iface.interfaceClass == 14 || iface.interfaceClass == 239) {
   567→                Log.i(TAG, "UVC device detected by interface class: ${iface.interfaceClass}")
   568→                return true
   569→            }
   570→        }
   571→
   572→        // Check by known Vendor IDs
   573→        val knownVids = intArrayOf(VENDOR_MACROSILICON, VENDOR_FUSHICAI, VENDOR_MAGEWELL)
   574→        if (knownVids.contains(device.vendorId)) {
   575→            Log.i(TAG, "UVC device detected by known VendorID: 0x${Integer.toHexString(device.vendorId)}")
   576→            return true
   577→        }
   578→
   579→        // Heuristic: class 239 subclass 2
   580→        if (device.interfaceCount >= 2 && device.vendorId != 0) {
   581→            for (i in 0 until device.interfaceCount) {
   582→                val iface = device.getInterface(i)
   583→                if (iface.interfaceClass == 239 && iface.interfaceSubclass == 2) {
   584→                    Log.i(TAG, "UVC device detected by class/subclass: 239/2")
   585→                    return true
   586→                }
   587→            }
   588→        }
   589→
   590→        return false
   591→    }

   615→    private fun requestUsbPermission(device: UsbDevice) {
   616→        try {
   617→            val usbManager = context.getSystemService(Context.USB_SERVICE) as? UsbManager
   618→            if (usbManager == null) {
   619→                Log.e(TAG, "UsbManager not available!")
   620→                return
   621→            }

   623→            if (usbManager.hasPermission(device)) {
   624→                Log.i(TAG, "USB permission already granted for ${device.deviceName}")
   625→                // On Android 14+, USBMonitor's onConnect may not fire due to
   626→                // registerReceiver() SecurityException. Open camera directly.
   627→                tryOpenUVCCameraDirect(device)
   628→                return
   629→            }

   631→            // FIX: Add FLAG_UPDATE_CURRENT to ensure PendingIntent contains correct

   636→            } else {
   637→                android.app.PendingIntent.FLAG_UPDATE_CURRENT

   640→            val pendingIntent = android.app.PendingIntent.getBroadcast(
   641→                context, device.deviceName.hashCode(),
   642→                Intent(ACTION_USB_PERMISSION).apply {
   643→                    putExtra(UsbManager.EXTRA_DEVICE, device)

   649→            Log.i(TAG, "USB permission requested for ${device.deviceName}")
   650→
   651→        } catch (e: Exception) {
   652→            Log.e(TAG, "Failed to request USB permission: ${e.message}")
   653→        }
   654→    }

   765→     * Set up the preview surface and start streaming.
   766→     * MUST be called on the MAIN thread (UI thread).

   768→    private fun setupPreviewSurface(camera: UVCCamera, tv: TextureView) {
   769→        try {
   770→            val surfaceTexture = tv.surfaceTexture
   771→            if (surfaceTexture == null) {
   772→                Log.e(TAG, "SurfaceTexture is null even though TextureView is available!")
   773→                return
   774→            }
   775→
   776→            // Create Surface from SurfaceTexture
   777→            val surface = Surface(surfaceTexture)
   778→            previewSurface = surface
   779→
   780→            // Use setPreviewDisplay(Surface) — the correct method for TextureView
   781→            // Do NOT also call setPreviewTexture() — they conflict
   782→            camera.setPreviewDisplay(surface)
   783→            Log.i(TAG, "★ setPreviewDisplay(Surface) called")
   784→
   785→            // START THE PREVIEW STREAM
   786→            camera.startPreview()
   787→            Log.i(TAG, "★★★ startPreview() called — video stream should now be visible ★★★")
   788→
   789→        } catch (e: Exception) {
   790→            Log.e(TAG, "Failed to setup preview surface: ${e.message}", e)
   791→            // Try alternative: setPreviewTexture instead
   792→            try {
   793→                val surfaceTexture = tv.surfaceTexture
   794→                if (surfaceTexture != null) {
   795→                    camera.setPreviewTexture(surfaceTexture)
   796→                    Log.i(TAG, "★ Fallback: setPreviewTexture() called")
   797→                    camera.startPreview()
   798→                    Log.i(TAG, "★★★ startPreview() called after fallback ★★★")
   799→                }
   800→            } catch (e2: Exception) {
   801→                Log.e(TAG, "Fallback setPreviewTexture also failed: ${e2.message}")
   802→            }
   803→        }
   804→    }

   951→    fun capturePhoto(onResult: (String?) -> Unit) {
   952→        val camera = uvcCamera
   953→        if (camera == null) {
   954→            Log.e(TAG, "capturePhoto: No UVCCamera instance")
   955→            onResult(null)
   956→            return
   957→        }
   958→
   959→        // Strategy 1: TextureView.getBitmap()
   960→        val tv = textureView
   961→        if (tv != null && tv.isAvailable) {
   962→            try {
   963→                val bitmap = tv.bitmap
   964→                if (bitmap != null && !bitmap.isRecycled) {
   965→                    if (!isBitmapBlack(bitmap)) {
   966→                        val outputStream = ByteArrayOutputStream()
   967→                        bitmap.compress(Bitmap.CompressFormat.JPEG, 95, outputStream)
   968→                        val bytes = outputStream.toByteArray()
   969→                        val base64 = Base64.encodeToString(bytes, Base64.NO_WRAP)
   970→                        val dataUrl = "data:image/jpeg;base64,$base64"
   971→                        outputStream.close()
   972→                        bitmap.recycle()
   973→
   974→                        Log.i(TAG, "Photo captured via TextureView bitmap (${dataUrl.length} chars)")
   975→                        onResult(dataUrl)
   976→                        return
   977→                    } else {
   978→                        Log.w(TAG, "TextureView bitmap is all black, trying IFrameCallback...")
   979→                        bitmap.recycle()
   980→                    }
   981→                }
   982→            } catch (e: Exception) {
   983→                Log.e(TAG, "TextureView bitmap capture failed: ${e.message}")
   984→            }
   985→        }
   986→
   987→        // Strategy 2: IFrameCallback
   988→        captureFromFrameCallback(camera, onResult)
   989→    }

   991→    private fun isBitmapBlack(bitmap: Bitmap): Boolean {
   992→        try {
   993→            val w = bitmap.width
   994→            val h = bitmap.height
   995→            val stepX = maxOf(1, w / 20)
   996→            val stepY = maxOf(1, h / 20)
   997→            var blackCount = 0
   998→            var totalCount = 0
   999→
  1000→            for (x in 0 until w step stepX) {
  1001→                for (y in 0 until h step stepY) {
  1002→                    val pixel = bitmap.getPixel(x, y)
  1003→                    totalCount++
  1004→                    val r = (pixel shr 16) and 0xFF
  1005→                    val g = (pixel shr 8) and 0xFF
  1006→                    val b = pixel and 0xFF
  1007→                    if (r + g + b < 30) {
  1008→                        blackCount++
  1009→                    }
  1010→                }
  1011→            }
  1012→
  1013→            val blackRatio = blackCount.toFloat() / totalCount.toFloat()
  1014→            return blackRatio > 0.95f
  1015→        } catch (e: Exception) {
  1016→            Log.e(TAG, "isBitmapBlack check failed: ${e.message}")
  1017→            return false
  1018→        }
  1019→    }

  1021→    private fun captureFromFrameCallback(camera: UVCCamera, callback: (String?) -> Unit) {
  1022→        // Single callback slot: fail any still-pending capture first so its
  1023→        // caller isn't leaked waiting forever.
  1024→        captureCallback?.invoke(null)
  1025→        captureCallback = callback
  1026→
  1027→        try {
  1028→            camera.setFrameCallback(object : IFrameCallback {
  1029→                override fun onFrame(frame: ByteBuffer) {
  1030→                    if (!frame.hasRemaining()) {
  1031→                        Log.e(TAG, "IFrameCallback: empty frame")
  1032→                        val cb = captureCallback
  1033→                        captureCallback = null
  1034→                        cb?.invoke(null)
  1035→                        return
  1036→                    }
  1037→
  1038→                    try {
  1039→                        val frameBytes = ByteArray(frame.remaining())
  1040→                        frame.get(frameBytes)
  1041→
  1042→                        // Use the ACTUAL negotiated size — after the 640x480
  1043→                        // fallback, decoding with 1280x720 corrupts the capture.
  1044→                        val yuvImage = android.graphics.YuvImage(
  1045→                            frameBytes, ImageFormat.NV21,
  1046→                            negotiatedWidth, negotiatedHeight, null
  1047→                        )
  1048→                        val outputStream = ByteArrayOutputStream()
  1049→                        yuvImage.compressToJpeg(
  1050→                            Rect(0, 0, negotiatedWidth, negotiatedHeight), 95, outputStream
  1051→                        )
  1052→                        val bytes = outputStream.toByteArray()
  1053→                        val base64 = Base64.encodeToString(bytes, Base64.NO_WRAP)
  1054→                        val dataUrl = "data:image/jpeg;base64,$base64"
  1055→                        outputStream.close()
  1056→
  1057→                        Log.i(TAG, "Photo captured via IFrameCallback (${dataUrl.length} chars)")
  1058→
  1059→                        val cb = captureCallback
  1060→                        captureCallback = null
  1061→                        cb?.invoke(dataUrl)
  1062→                    } catch (e: Exception) {
  1063→                        Log.e(TAG, "Error converting IFrameCallback frame: ${e.message}")
  1064→                        val cb = captureCallback
  1065→                        captureCallback = null
  1066→                        cb?.invoke(null)
  1067→                    }
  1068→
  1069→                    // Remove callback after one frame
  1070→                    try {
  1071→                        camera.setFrameCallback(null, 0)
  1072→                    } catch (_: Exception) {}
  1073→                }
  1074→            }, UVCCamera.PIXEL_FORMAT_NV21)
  1075→
  1076→            // Timeout — guarded to the specific callback instance this timeout
  1077→            // belongs to, so the timeout of an OLD capture can't kill a NEW
  1078→            // capture's callback (captureCallback may have been replaced).
  1079→            backgroundHandler?.postDelayed({
  1080→                if (captureCallback === callback) {
  1081→                    Log.w(TAG, "IFrameCallback timed out")
  1082→                    try { camera.setFrameCallback(null, 0) } catch (_: Exception) {}
  1083→
  1084→                    val tv = textureView
  1085→                    if (tv != null && tv.isAvailable) {
  1086→                        try {
  1087→                            val bitmap = tv.bitmap
  1088→                            if (bitmap != null && !bitmap.isRecycled) {
  1089→                                val outputStream = ByteArrayOutputStream()
  1090→                                bitmap.compress(Bitmap.CompressFormat.JPEG, 95, outputStream)
  1091→                                val bytes = outputStream.toByteArray()
  1092→                                val base64 = Base64.encodeToString(bytes, Base64.NO_WRAP)
  1093→                                val dataUrl = "data:image/jpeg;base64,$base64"
  1094→                                outputStream.close()
  1095→                                bitmap.recycle()
  1096→
  1097→                                captureCallback = null
  1098→                                callback(dataUrl)
  1099→                                return@postDelayed
  1100→                            }
  1101→                        } catch (_: Exception) {}
  1102→                    }
  1103→
  1104→                    captureCallback = null
  1105→                    callback(null)
  1106→                }
  1107→            }, 3000)
  1108→
  1109→        } catch (e: Exception) {
  1110→            Log.e(TAG, "IFrameCallback setup failed: ${e.message}")
  1111→            captureCallback = null
  1112→            callback(null)
  1113→        }
  1114→    }
